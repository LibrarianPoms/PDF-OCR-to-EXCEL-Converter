Sub OpenPACInterne()
    Dim wordApp As Object
    Dim wordDoc As Object
    Dim docPath As String
    
    ' Path to the Word document (same folder as this Excel file)
    docPath = ThisWorkbook.Path & "\Exemple Pac Interne (2).doc"
    
    On Error Resume Next
    
    ' Try to get existing Word application or create new one
    Set wordApp = GetObject(, "Word.Application")
    If wordApp Is Nothing Then
        Set wordApp = CreateObject("Word.Application")
    End If
    
    ' Make Word visible
    wordApp.Visible = True
    
    ' Check if document exists and open it
    If Dir(docPath) <> "" Then
        Set wordDoc = wordApp.Documents.Open(docPath)
        MsgBox "PAC Interne document opened successfully for editing!", vbInformation, "PAC Interne"
    Else
        MsgBox "Document 'Exemple Pac Interne (2).doc' not found in the same folder!", vbExclamation, "File Not Found"
    End If
    
    On Error GoTo 0
End Sub
